using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Events;

[System.Serializable]
public class CampusNarrativeState { public List<string> played=new List<string>(); }
public enum CampusVoice { Inner, Broadcast, NPC, Feedback }
[System.Serializable]
public class CampusAudioCue { public string eventId; public AudioClip clip; }

/// <summary>One subtitle surface; persistent one-shot events and repeatable investigation/dialogue.</summary>
public class CampusNarrative : MonoBehaviour
{
    public AudioSource voice, ambience;
    public CampusAudioCue[] cues=new CampusAudioCue[0];
    public UnityEvent<string> onLine=new UnityEvent<string>();
    struct Line { public string text; public float seconds; public CampusVoice channel; public AudioClip clip; }
    readonly Queue<Line> queue=new Queue<Line>();
    Line current;
    float until;
    public string CurrentText { get { return until>Time.time?current.text:""; } }
    public int PendingCount { get { return queue.Count; } }
    public bool Emit(string id,CampusVoice channel,string text,float seconds=7,bool once=true)
    {
        var state=GameProgress.Ensure().narrative;
        if(once && !string.IsNullOrEmpty(id)) { if(state.played.Contains(id))return false;state.played.Add(id); }
        var line=new Line{text=text,seconds=seconds,channel=channel};
        foreach(var cue in cues)if(cue.eventId==id){line.clip=cue.clip;break;}
        if(until>Time.time){if(queue.Count<6)queue.Enqueue(line);else {queue.Clear();Begin(line);}}else Begin(line);
        return true;
    }
    void Begin(Line line){current=line;until=Time.time+line.seconds;if(voice!=null && line.clip!=null)voice.PlayOneShot(line.clip);onLine.Invoke(line.text);}
    public void Skip(){if(voice!=null)voice.Stop();until=0;if(queue.Count>0)Begin(queue.Dequeue());}
    public void Clear(){queue.Clear();until=0;if(voice!=null)voice.Stop();}
    void Start(){GameProgress.Ensure().SaveApplied+=Clear;}
    void OnDestroy(){if(GameProgress.Instance!=null)GameProgress.Instance.SaveApplied-=Clear;}
    void Update()
    {
        if(Time.timeScale==0)return;
        if(until>0 && Time.time>=until){until=0;if(queue.Count>0)Begin(queue.Dequeue());}
        var exam=GetComponent<CampusExamChapter>();
        if(Input.GetKeyDown(KeyCode.Backspace) && (exam==null || !exam.KeypadOpen))Skip();
    }
    void OnGUI()
    {
        if(CurrentText.Length==0 || (PauseMenu.Instance!=null && PauseMenu.Instance.IsOpen) || CampusJournal.Opened)return;
        var exam=GetComponent<CampusExamChapter>();if(exam!=null && exam.KeypadOpen)return;
        float scale=Mathf.Min(Screen.width/1280f,Screen.height/720f);var matrix=GUI.matrix;GUI.matrix=Matrix4x4.Scale(Vector3.one*scale);
        float width=Screen.width/scale,height=Screen.height/scale;
        string label=current.channel==CampusVoice.Inner?"内心独白":current.channel==CampusVoice.Broadcast?"校园广播":current.channel==CampusVoice.NPC?"对话":"调查与任务";
        var style=new GUIStyle(GUI.skin.label){font=Resources.Load<Font>("UIFont"),fontSize=19,wordWrap=true};
        GUI.Box(new Rect(100,height-260,width-200,112),"");GUI.Label(new Rect(118,height-254,width-236,102),label+" · 退格键跳过\n"+current.text,style);GUI.matrix=matrix;
    }
}
